cmake_minimum_required(VERSION 3.24)
project(iiSocietySyncConsumer LANGUAGES CXX)
enable_testing()
find_package(iiSocietySync 0.4.0 CONFIG REQUIRED)
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Test)
set(CMAKE_AUTOMOC ON)
set(IISOCIETYSYNC_TEST_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}" CACHE STRING "Test data directory as seen by the target runtime")
get_target_property(sync_dependencies iiSocietySync::iiSocietySync INTERFACE_LINK_LIBRARIES)
if(sync_dependencies MATCHES "iiSocietyHelper")
    message(FATAL_ERROR "Device synchronization must not depend on local Helper IPC")
endif()
add_executable(iiSocietySync_consumer ../hello_world.cpp)
target_link_libraries(iiSocietySync_consumer PRIVATE iiSocietySync::iiSocietySync)
set_target_properties(iiSocietySync_consumer PROPERTIES CXX_EXTENSIONS OFF)
add_test(NAME iiSocietySync.installed_consumer COMMAND iiSocietySync_consumer)
foreach(test replica synchronizer controller)
    add_executable(iiSocietySync_installed_${test} ../${test}.cpp)
    target_link_libraries(iiSocietySync_installed_${test} PRIVATE iiSocietySync::iiSocietySync Qt6::Test)
    target_compile_definitions(iiSocietySync_installed_${test} PRIVATE SYNC_TEST_DIRECTORY="${IISOCIETYSYNC_TEST_DIRECTORY}")
    add_test(NAME iiSocietySync.installed_${test} COMMAND iiSocietySync_installed_${test})
    set_tests_properties(iiSocietySync.installed_${test} PROPERTIES TIMEOUT 90)
endforeach()
